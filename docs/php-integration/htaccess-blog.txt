# ── Blog URL routing for urologybymanmeet.com ─────────────────────────────────
#
# WHERE THIS GOES:
#   Create a new file: /blog/.htaccess  (inside the /blog/ subdirectory)
#   If /blog/ does not exist as a real directory, create it on the server.
#
# WHAT IT DOES:
#   /blog/                                → index.php (blog-list.php renamed)
#   /blog/?page=2                         → index.php?page=2
#   /blog/kidney-stone-treatment/         → post.php?slug=kidney-stone-treatment
#   /blog/kidney-stone-treatment          → 301 redirect → /blog/kidney-stone-treatment/
#
# HOW TO TEST:
#   After uploading, visit: urologybymanmeet.com/blog/
#   Then visit a real slug: urologybymanmeet.com/blog/urinary-incontinence-treatment-in-lucknow/

Options -Indexes
RewriteEngine On
RewriteBase /blog/

# ── Trailing slash enforcement ────────────────────────────────────────────────
# Redirect /blog/slug to /blog/slug/ (canonical URL consistency)
RewriteCond %{REQUEST_FILENAME} !-f
RewriteCond %{REQUEST_FILENAME} !-d
RewriteCond %{REQUEST_URI} !/$
RewriteRule ^([a-z0-9-]+)$ /blog/$1/ [R=301,L]

# ── Route /blog/SLUG/ to post.php ─────────────────────────────────────────────
RewriteCond %{REQUEST_FILENAME} !-f
RewriteCond %{REQUEST_FILENAME} !-d
RewriteRule ^([a-z0-9-]+)/$ post.php?slug=$1 [QSA,L]
